import heapq

def plan_order(graph):
    if not graph:
        return []

    for task, deps in graph.items():
        for dep in deps:
            if dep not in graph:
                raise ValueError(f"dependency {dep!r} for task {task!r} is absent from the graph")

    indegree = {task: 0 for task in graph}
    dependents = {task: [] for task in graph}

    for task in sorted(graph):
        unique_deps = set(graph[task])
        indegree[task] = len(unique_deps)
        for dep in sorted(unique_deps):
            dependents[dep].append(task)

    ready = [task for task in sorted(graph) if indegree[task] == 0]
    heapq.heapify(ready)
    order = []

    while ready:
        task = heapq.heappop(ready)
        order.append(task)
        for dependent in dependents[task]:
            indegree[dependent] -= 1
            if indegree[dependent] == 0:
                heapq.heappush(ready, dependent)

    if len(order) != len(graph):
        raise ValueError("task graph contains a cycle")

    return order
